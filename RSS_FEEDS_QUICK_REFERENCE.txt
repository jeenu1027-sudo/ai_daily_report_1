import requests
import xml.etree.ElementTree as ET

def fetch_korean_ai_news():
    """한국 AI 뉴스 수집"""
    sources = [
        'https://www.venturesquare.net/feed',
        'https://news.google.com/rss/search?q=AI%20%ED%95%9C%EA%B5%AD&hl=ko&gl=KR&ceid=KR:ko',
        'https://zdnet.co.kr/feed/',
    ]

    all_news = []

    for url in sources:
        try:
            headers = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)'}
            response = requests.get(url, headers=headers, timeout=10)

            root = ET.fromstring(response.content)
            items = root.findall('.//item') or root.findall('.//{http://www.w3.org/2005/Atom}entry')

            for item in items[:5]:
                title_elem = item.find('title') or item.find('{http://www.w3.org/2005/Atom}title')
                if title_elem is not None and title_elem.text:
                    all_news.append(title_elem.text)

        except Exception as e:
            print(f"Error fetching {url}: {e}")

    return all_news

# 사용 예
news = fetch_korean_ai_news()
for i, item in enumerate(news[:10], 1):
    print(f"{i}. {item}")
